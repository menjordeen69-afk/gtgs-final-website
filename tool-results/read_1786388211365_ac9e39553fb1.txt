    49→function formatFileSize(bytes: number): string {
    50→  if (bytes < 1024) return bytes + ' B'
    51→  if (bytes < 1048576) return (bytes / 1024).toFixed(1) + ' KB'
    52→  return (bytes / 1048576).toFixed(1) + ' MB'
    53→}

    67→  async function login() {
    68→    setLoading(true)
    69→    setError('')
    70→    try {
    71→      const res = await fetch('/api/admin', {
    72→        headers: { 'Authorization': `Bearer ${password}` },
    73→      })
    74→      if (res.ok) {
    75→        const json = await res.json()
    76→        setApplications(json.applications || [])
    77→        setAuthenticated(true)
    78→      } else if (res.status === 429) {
    79→        setError('Too many failed attempts. Locked for 15 minutes.')
    80→      } else {
    81→        const json = await res.json().catch(() => ({}))
    82→        const remaining = (json as Record<string, unknown>).remainingAttempts
    83→        setError(remaining ? `Incorrect password. ${remaining} attempt${Number(remaining) > 1 ? 's' : ''} remaining.` : 'Incorrect password')
    84→      }
    85→    } catch {
    86→      setError('Connection failed')
    87→    } finally {
    88→      setLoading(false)
    89→    }
    90→  }
    91→
    92→  async function refreshData() {
    93→    try {
    94→      const res = await fetch('/api/admin', {
    95→        headers: { 'Authorization': `Bearer ${password}` },
    96→      })
    97→      if (res.ok) {
    98→        const json = await res.json()
    99→        setApplications(json.applications || [])
   100→      }
   101→    } catch { /* ignore */ }
   102→  }

   104→  async function updateStatus(id: string, status: string) {
   105→    const res = await fetch('/api/admin', {
   106→      method: 'PATCH',
   107→      headers: {
   108→        'Content-Type': 'application/json',
   109→        'Authorization': `Bearer ${password}`,
   110→      },
   111→      body: JSON.stringify({ id, status }),
   112→    })

   114→      setApplications((prev) => prev.map(a => a.id === id ? { ...a, status } : a))
   115→      if (selectedApp?.id === id) setSelectedApp((prev) => prev ? { ...prev, status } : null)
   116→    }
   117→  }

   119→  async function deleteApplication(id: string) {
   120→    const res = await fetch(`/api/admin?id=${id}`, {
   121→      method: 'DELETE',
   122→      headers: { 'Authorization': `Bearer ${password}` },
   123→    })
   124→    if (res.ok) {
   125→      setApplications((prev) => prev.filter(a => a.id !== id))
   126→      if (selectedApp?.id === id) setSelectedApp(null)
   127→      setDeleteConfirm(null)
   128→    }

   131→  async function downloadDocument(savedPath: string, fileName: string) {
   132→    try {
   133→      const res = await fetch(`/api/documents?file=${encodeURIComponent(savedPath)}`, {
   134→        headers: { 'Authorization': `Bearer ${password}` },
   135→      })
   136→      if (!res.ok) return
   137→      const blob = await res.blob()
   138→      const url = URL.createObjectURL(blob)
   139→      const a = document.createElement('a')
   140→      a.href = url
   141→      a.download = fileName
   142→      a.click()
   143→      URL.revokeObjectURL(url)
   144→    } catch { /* ignore */ }
   145→  }
   146→
   147→  function viewDocument(savedPath: string) {
   148→    const url = `/api/documents?file=${encodeURIComponent(savedPath)}&token=${password}`
   149→    window.open(url, '_blank')
